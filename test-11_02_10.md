### Q8. Pattern Breaker — Self-Modifying Loop [5 Marks]

Trace the program carefully and write the **exact output**.

```c
#include <stdio.h>

int main()
{
    int x = 2;

    for (int i = 1; i <= 5; i++)
    {
        x += i;

        if (x % 3 == 0)
            i++;

        if (i % 2 == 0)
            x -= 2;
        else
            x += 1;

        printf("%d ", x);
    }

    return 0;
}
```

### Q9. Pattern Breaker — Array Mutation + Changing Bounds [5 Marks]

Trace the program and write the **exact output**.

```c
#include <stdio.h>

int main()
{
    int a[] = {2, 4, 6, 8};
    int total = 0;

    for (int i = 0; i < 4; i++)
    {
        for (int j = i; j < 4; j++)
        {
            total += a[j];

            if (total % 3 == 0)
            {
                a[j] = a[j] / 2;
                total -= a[j];
            }
            else
            {
                a[j] = a[j] + i;
                total += a[j];
            }

            printf("%d ", total);
        }
    }

    return 0;
}
```
