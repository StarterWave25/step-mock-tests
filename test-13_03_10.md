### Q8. Pattern Breaker — `continue` Changes the Effective Sequence [5 Marks]

Trace the following program and write the **exact output**.

```c
#include <stdio.h>

int main()
{
    int x = 10;

    for (int i = 0; i < 5; i++)
    {
        x -= i;

        if (x % 3 == 0)
        {
            x += 2;
            continue;
        }

        x = x / 2 + i;

        if (x % 2 == 0)
            x += 3;
        else
            x -= 1;

        printf("%d ", x);
    }

    return 0;
}
```

### Q9. Pattern Breaker — Array Mutation + Cross-Index Dependency [5 Marks]

Trace the program and write the **exact output**.

```c
#include <stdio.h>

int main()
{
    int a[] = {2, 3, 4, 5};
    int x = 1;

    for (int i = 0; i < 4; i++)
    {
        x = x + a[i];

        if (x % 2 == 0)
            a[i] = a[i] + x / 2;
        else
            a[i] = a[i] - 1;

        if (i > 0)
            x = x - a[i - 1];

        printf("%d ", x);
    }

    return 0;
}
```
