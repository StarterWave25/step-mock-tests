### 8. [5 Marks]

```c
#include <stdio.h>

int func(int n)
{
    return n % 4;
}

int main(void)
{
    int arr[] = {11, 16, 19, 22, 27};

    for (int i = 0; i < 5; i++)
    {
        if (func(arr[i]) == 0 || func(arr[i]) == 3)
        {
            printf("%d ", arr[i] - 1);
        }
    }

    return 0;
}
```

### 9. [5 Marks]

Trace the following program **step by step** and write the exact output.

```c
#include <stdio.h>

int main(void)
{
    int total = 0;

    for (int i = 0; i < 3; i++)
    {
        for (int j = 1; j <= 2; j++)
        {
            if (i + j <= 2)
            {
                total += i + j;
                printf("%d ", total);
            }
            else
            {
                printf("%d ", i + j);
            }
        }
    }

    return 0;
}
```
