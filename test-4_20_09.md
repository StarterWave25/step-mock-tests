## Section C: Code Tracing

**2 × 5 = 10 Marks**

### 8. [5 Marks]

Trace the following C program **step by step** and write the **exact output**.

```c
#include <stdio.h>

int func(int n)
{
    return n % 4;
}

int main(void)
{
    int arr[] = {13, 18, 23, 29, 34};

    for (int i = 0; i < 5; i++)
    {
        int x = func(arr[i]);

        if (x == 0)
        {
            printf("%d ", arr[i] / 2);
        }
        else if (x % 2 == 0)
        {
            printf("%d ", arr[i] - x);
        }
        else
        {
            printf("%d ", arr[i] + x);
        }
    }

    return 0;
}
```

### 9. [5 Marks]

```c

{
    int total = 1;

    for (int i = 0; i < 3; i++)
    {
        for (int j = 0; j < 3; j++)
        {
            if ((i + j) % 2 == 0)
            {
                total += i + j;
            }
            else
            {
                total *= (j + 1);
            }

            printf("%d ", total);

            if (total % 3 == 0)
            {
                printf("%d ", i - j);
            }
        }
    }

    return 0;
}
```
